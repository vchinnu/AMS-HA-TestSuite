//find the most recent records for each cluster node, identify the master (status = DC) and return the correlation ID from that node only
let master = materialize(Prometheus_HaClusterExporter_CL
    | where TimeGenerated > ago(40min)
    | where value_d == 1
    | extend node_status=parse_json(labels_s)
    | where node_status['status'] == 'dc'
    | where sid_s == '{selected_sid}'
    | where clusterName_s == '{selected_clustername}'
    | where tostring(node_status['node']) == hostname_s
    | summarize arg_max(TimeGenerated, correlation_id_g, value_d) by sid_s, clusterName_s, hostname_s
    | top 1 by TimeGenerated
    | project correlation_id_g);
// find all resources which have their role as stopped
let stoppedResources = (Prometheus_HaClusterExporter_CL
    | where correlation_id_g in (master)
    | where name_s == 'ha_cluster_pacemaker_resources'
    | extend resources = parse_json(labels_s)
    | where isnotnull(resources['agent'])
    | where resources['role'] == 'stopped'
    | project
        correlation_id_g,
        name_s,
        value_d,
        resource_agent = tostring(resources['agent']),
        resource_resource = tostring(resources['resource']),
        resource_node = tostring(resources['node']),
        resource_managed = tostring(resources['managed']),
        resource_role= tostring(resources['role']),
        resource_status=''
    | summarize status_managed = countif(tolower(resource_managed) == 'true')
        by
        correlation_id_g,
        name_s,
        value_d,
        resource_agent,
        resource_resource,
        resource_node,
        resource_role,
        resource_status
    | summarize resource_managed = case(sum(status_managed) > 0, 'true', 'false')
        by
        correlation_id_g,
        name_s,
        value_d,
        resource_agent,
        resource_resource,
        resource_node,
        resource_role,
        resource_status);
//find all pacemaker resources and their status from dc metrics
Prometheus_HaClusterExporter_CL
| where correlation_id_g in (master)
| where name_s == 'ha_cluster_pacemaker_resources'
| where value_d == 1
| extend resources = parse_json(labels_s)
| where isnotnull(resources['agent'])
| where strlen(resources['node']) > 0
| project
    correlation_id_g,
    name_s,
    value_d,
    resource_agent = resources['agent'],
    resource_resource = resources['resource'],
    resource_node = resources['node'],
    resource_managed = resources['managed'],
    resource_role= resources['role'],
    resource_status=resources['status']
//| summarize count() by correlation_id_g, name_s, value_d,tostring(resource_agent),tostring(resource_resource), tostring(resource_node),tostring(resource_managed)
| summarize
    status_managed = countif(tolower(resource_managed) == 'true'),
    status_active = countif(tolower(resource_status) == 'active') + countif(tolower(resource_role) == 'started' and resource_managed == 'true') + countif(tolower(resource_role) == 'master' and resource_managed == 'true') + countif(tolower(resource_role) == 'slave' and resource_managed == 'true'),
    status_orphaned = countif(tolower(resource_status) == 'orphaned'),
    status_blocked = countif(tolower(resource_status) == 'blocked'),
    status_failed = countif(tolower(resource_status) == 'failed'),
    status_failure_ignored = countif(tolower(resource_status) == 'failure_ignored'),
    status_role_master = countif(tolower(resource_role) == 'master'),
    status_role_slave = countif(tolower(resource_role) == 'slave')
    by
    correlation_id_g,
    name_s,
    value_d,
    tostring(resource_agent),
    tostring(resource_resource),
    tostring(resource_node)
| summarize
    resource_status = case(sum(status_failure_ignored) > 0, 'failure_ignored', sum(status_failed) > 0, 'failed', sum(status_blocked) > 0, 'blocked', sum(status_orphaned) > 0, 'orphaned', sum(status_active) > 0, 'active', 'inactive'),
    resource_role = case(sum(status_role_master) > 0, 'primary', sum(status_role_slave) > 0, 'secondary', ''),
    resource_managed = case(sum(status_managed) > 0, 'true', 'false')
    by
    correlation_id_g,
    name_s,
    value_d,
    tostring(resource_agent),
    tostring(resource_resource),
    tostring(resource_node)
| union stoppedResources
